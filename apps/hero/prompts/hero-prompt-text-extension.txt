                settings.minimumHeight ? "hero-min-height" : ""
            ].filter(Boolean).join(" ");

                ? "hero-h1-serif"
                : "hero-h1-sans";

            // Select H2 font class.
            const h2FontClass = settings.h2SerifFont
                ? "hero-h2-serif"
                : "hero-h2-sans";

            // Add H2 only when enabled.
            let subheadingHTML = "";

            if (settings.showSubheading) {

                subheadingHTML = `
        <h2 class="${h2FontClass}">${subheadingText}</h2>
                `.trim();

            }

            // Return semantic hero HTML.
            return `
<header class="${heroClasses}">
    <div class="hero-content">
        <h1 class="${h1FontClass}">${headingText}</h1>
        ${subheadingHTML}
    </div>
</header>
            `.trim();

        }

        function updateFallbackHTML() {

            fallbackTextarea.value = generateCompleteHTML();

        }


        // ==========================================
        // 13. HANDLE PLAIN TEXT PASTING
        // ==========================================

        function handlePlainTextPaste(event) {

            event.preventDefault();

            // Get clipboard text without HTML formatting.
            const plainText = event.clipboardData.getData("text/plain");

            const selection = window.getSelection();

            if (!selection || selection.rangeCount === 0) {
                return;
            }

            const range = selection.getRangeAt(0);

            // Remove the selected text.
            range.deleteContents();

            // Insert plain text.
            const textNode = document.createTextNode(plainText);

            range.insertNode(textNode);

            // Position the cursor after inserted text.
            range.setStartAfter(textNode);

            range.collapse(true);

            selection.removeAllRanges();

            selection.addRange(range);

            // Update exported HTML.
            updateFallbackHTML();

        }


        // ==========================================
        // 14. COPY HTML TO CLIPBOARD
        // ==========================================

        async function copyHTML() {

            // Generate the latest HTML.
            const html = generateCompleteHTML();

            // Update the manual copy textarea.
            fallbackTextarea.value = html;

            statusMessage.textContent = "";

            try {

                // Check Clipboard API availability.
                if (!navigator.clipboard || !navigator.clipboard.writeText) {

                    throw new Error("Clipboard API unavailable");

                }

                // Copy the generated HTML.
                await navigator.clipboard.writeText(html);

                // Display success only after copying succeeds.
                statusMessage.textContent =
                    "HTML copied—paste it into VS Code.";

                fallbackContainer.hidden = true;

            } catch (error) {

                // Show manual copy option.
                fallbackTextarea.value = html;

                fallbackContainer.hidden = false;

                statusMessage.textContent =
                    "Automatic copying is unavailable. Use the textarea below to copy your HTML.";

            }

        }

        heroSubheading.addEventListener("input", updateFallbackHTML);


        // Prevent rich-text formatting when pasting.
        heroHeading.addEventListener("paste", handlePlainTextPaste);

        heroSubheading.addEventListener("paste", handlePlainTextPaste);


        // Copy button.
        copyButton.addEventListener("click", copyHTML);
